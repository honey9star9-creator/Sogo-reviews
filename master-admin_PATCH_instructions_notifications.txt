STEP D — Add JS logic
--------------------------------------------------------------
Inside the <script type="module"> block, add this whole section
right AFTER the "PAYMENTS" section (after the
`paymentsSearchInputElem.oninput = ...` line is a good spot):

  // ---------------- NOTIFICATIONS ----------------
  let adminNotifOpen = false;

  function adminNotifColor(type) {
    const incoming = ['package_joined', 'payment_proof_uploaded', 'deposit_requested', 'review_submitted', 'withdrawal_requested', 'topup_requested', 'password_reset_requested'];
    return incoming.includes(type) ? '#6366f1' : '#0f172a';
  }

  async function refreshAdminNotifications() {
    const authApi = await waitForSogoAuth;
    if (!authApi || !currentAdminSession) return;
    const [list, unread] = await Promise.all([
      authApi.getMyNotifications(currentAdminSession.uid),
      authApi.getUnreadNotificationCount(currentAdminSession.uid)
    ]);
    const badge = document.getElementById('adminNotifBadge');
    if (badge) {
      if (unread > 0) { badge.style.display = 'flex'; badge.textContent = unread > 9 ? '9+' : unread; }
      else badge.style.display = 'none';
    }
    const listEl = document.getElementById('adminNotifList');
    if (!listEl) return;
    if (!list.length) {
      listEl.innerHTML = '<div style="padding:30px 18px;text-align:center;color:var(--text-muted);font-size:0.75rem;">No notifications yet.</div>';
      return;
    }
    listEl.innerHTML = list.map(n => `
      <div class="notif-item ${n.read ? '' : 'unread'}">
        <div class="notif-item-title" style="color:${adminNotifColor(n.type)};">${escapeHtml(n.title)}</div>
        <div class="notif-item-desc">${escapeHtml(n.message)}</div>
        <div class="notif-item-time">${formatDate(n.created_at)} ${new Date(n.created_at).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}</div>
      </div>
    `).join('');
  }

  window.toggleAdminNotifDropdown = async function (force) {
    const dropdown = document.getElementById('adminNotifDropdown');
    if (!dropdown) return;
    adminNotifOpen = (typeof force === 'boolean') ? force : !adminNotifOpen;
    dropdown.style.display = adminNotifOpen ? 'flex' : 'none';
    if (adminNotifOpen) {
      await refreshAdminNotifications();
      const authApi = await waitForSogoAuth;
      if (authApi && currentAdminSession) {
        await authApi.markAllNotificationsRead(currentAdminSession.uid);
        const badge = document.getElementById('adminNotifBadge');
        if (badge) badge.style.display = 'none';
      }
    }
  };

  document.addEventListener('click', (e) => {
    const wrap = document.querySelector('.notif-bell-wrap');
    if (wrap && !wrap.contains(e.target)) window.toggleAdminNotifDropdown(false);
  });

  // ---------------- MANUAL NOTIFY (Users table -> Notify link) ----------------
  window.adminOpenNotifyModal = function (uid) {
    const user = allUsersList.find(u => u.uid === uid);
    if (!user) return;

    const title = document.getElementById('adminModalTitle');
    const body = document.getElementById('adminModalBody');
    if (title) title.textContent = 'Send notification — ' + (user.name || user.email);

    if (body) {
      body.innerHTML = `
        <div class="form-group" style="margin-bottom:14px;">
          <label class="form-label">Title</label>
          <input type="text" class="form-input" id="notifyTitleInput" placeholder="e.g. Account update">
        </div>
        <div class="form-group" style="margin-bottom:14px;">
          <label class="form-label">Message</label>
          <textarea class="form-textarea" id="notifyMessageInput" placeholder="Type your message to this user..."></textarea>
        </div>
        <button class="btn-purple" style="width:100%;" onclick="adminSendManualNotification('${uid}')">Send notification</button>
        <p id="notifySendMsg" style="font-size:0.75rem; margin-top:10px; display:none;"></p>
      `;
    }

    openAdminModal();
  };

  window.adminSendManualNotification = async function (uid) {
    const authApi = await waitForSogoAuth;
    const titleInput = document.getElementById('notifyTitleInput');
    const msgInput = document.getElementById('notifyMessageInput');
    const feedback = document.getElementById('notifySendMsg');
    const title = titleInput ? titleInput.value.trim() : '';
    const message = msgInput ? msgInput.value.trim() : '';

    const result = await authApi.sendManualNotification(uid, title, message);
    if (feedback) {
      feedback.style.display = 'block';
      feedback.style.color = result.ok ? '#10b981' : '#dc2626';
      feedback.textContent = result.ok ? 'Notification sent successfully!' : (result.message || 'Could not send notification.');
    }
    if (result.ok) {
      if (titleInput) titleInput.value = '';
      if (msgInput) msgInput.value = '';
    }
  };
