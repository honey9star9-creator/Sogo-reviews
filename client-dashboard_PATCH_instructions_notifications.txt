STEP C — Add JS logic
--------------------------------------------------------------
Inside the main <script> block, paste this anywhere near the other
helper functions (for example right after the
"loadMyDepositsForPackage" / "findNextLock" functions):

    function notifColor(type) {
      const positive = ['topup_approved', 'products_unlocked', 'review_approved', 'package_activated'];
      const negative = ['review_rejected', 'deposit_rejected', 'withdrawal_rejected', 'package_deactivated', 'package_rejected', 'topup_rejected'];
      if (positive.includes(type)) return '#2ecc71';
      if (negative.includes(type)) return '#e74c3c';
      return '#e5b373';
    }

    function formatNotifTime(iso) {
      if (!iso) return '';
      const d = new Date(iso);
      if (isNaN(d)) return '';
      return d.toLocaleDateString() + ', ' + d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });
    }

    let notifDropdownOpen = false;

    async function refreshNotifications() {
      if (!currentSession || !window.SogoAuth) return;
      const [list, unread] = await Promise.all([
        SogoAuth.getMyNotifications(currentSession.uid),
        SogoAuth.getUnreadNotificationCount(currentSession.uid)
      ]);
      const badge = document.getElementById('notifBadge');
      if (badge) {
        if (unread > 0) { badge.style.display = 'flex'; badge.textContent = unread > 9 ? '9+' : unread; }
        else badge.style.display = 'none';
      }
      const listEl = document.getElementById('notifList');
      if (!listEl) return;
      if (!list.length) {
        listEl.innerHTML = '<div style="padding:30px 20px;text-align:center;color:var(--text-muted);font-size:12px;">No notifications yet.</div>';
        return;
      }
      listEl.innerHTML = list.map(n => `
        <div class="notif-item ${n.read ? '' : 'unread'}">
          <div class="notif-item-title" style="color:${notifColor(n.type)};">${escapeHtmlC(n.title)}</div>
          <div class="notif-item-desc">${escapeHtmlC(n.message)}</div>
          <div class="notif-item-time">${formatNotifTime(n.created_at)}</div>
        </div>
      `).join('');
    }

    async function toggleNotifDropdown(force) {
      const dropdown = document.getElementById('notifDropdown');
      if (!dropdown) return;
      notifDropdownOpen = (typeof force === 'boolean') ? force : !notifDropdownOpen;
      dropdown.style.display = notifDropdownOpen ? 'flex' : 'none';
      if (notifDropdownOpen) {
        await refreshNotifications();
        if (currentSession && window.SogoAuth) await SogoAuth.markAllNotificationsRead(currentSession.uid);
        const badge = document.getElementById('notifBadge');
        if (badge) badge.style.display = 'none';
      }
    }

    document.addEventListener('click', (e) => {
      const wrap = document.querySelector('.notif-bell-wrap');
      if (wrap && !wrap.contains(e.target)) toggleNotifDropdown(false);
    });
