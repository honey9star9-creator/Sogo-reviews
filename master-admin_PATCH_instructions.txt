  // ---------------- PAYMENTS ----------------
  let allPaymentJoins = [];
  let allPaymentDeposits = [];
  let paymentsFilter = 'all';

  window.setPaymentsFilter = function (filter, el) {
    paymentsFilter = filter;
    document.querySelectorAll('#paymentsFilterTabs .filter-tab').forEach(b => b.classList.remove('active'));
    if (el) el.classList.add('active');
    renderPaymentsList();
  };

  async function refreshPayments() {
    if (!window.SogoPackages) return;
    const [subs, deposits] = await Promise.all([
      SogoPackages.getAllUserPackages(),
      SogoPackages.getAllPackageDeposits ? SogoPackages.getAllPackageDeposits() : Promise.resolve([])
    ]);
    allPaymentJoins = (subs || []).filter(s => s.status === 'pending_payment' || s.status === 'pending_verification');
    allPaymentDeposits = (deposits || []).filter(d => d.status === 'pending');
    renderPaymentsList();
  }

  function renderPaymentsList() {
    const container = document.getElementById('paymentsListContainer');
    if (!container) return;

    const q = ((document.getElementById('paymentsSearchInput') || {}).value || '').toLowerCase();

    let joins = paymentsFilter === 'deposit' ? [] : allPaymentJoins;
    let deposits = paymentsFilter === 'join' ? [] : allPaymentDeposits;

    if (q) {
      joins = joins.filter(s => (s.userEmail || '').toLowerCase().includes(q) || (s.packageName || '').toLowerCase().includes(q));
      deposits = deposits.filter(d => (d.user_email || '').toLowerCase().includes(q));
    }

    if (!joins.length && !deposits.length) {
      container.innerHTML = '<div class="card" style="text-align:center; color:var(--text-muted); padding:40px;">No pending payments.</div>';
      return;
    }

    let html = '';

    joins.forEach(s => {
      const userName = (allUsersList.find(u => u.uid === s.userId) || {}).name || (s.userEmail || '').split('@')[0];
      const proofCell = s.paymentProofUrl
        ? `<a href="${escapeHtml(s.paymentProofUrl)}" target="_blank" style="color:inherit; text-decoration:underline;">View<br>proof</a>`
        : 'No<br>proof';
      html += `
        <div class="payment-req-card">
          <div class="payment-user-info">
            <div class="proof-badge">${proofCell}</div>
            <div class="payment-details">
              <h4>${escapeHtml(userName)}</h4>
              <p>${escapeHtml(s.userEmail || '')}</p>
              <div class="payment-sub">${escapeHtml(s.packageName || '')} - $${Number(s.price || 0).toFixed(2)} · Package join payment</div>
              <p style="margin-top: 2px;">${formatDate(s.joinedAt)}</p>
            </div>
          </div>
          <div class="payment-actions">
            <button class="btn-verify" onclick="adminVerifySubscription('${s.id}')">Verify & Activate</button>
            <button class="btn-reject" onclick="adminRejectJoin('${s.id}')">Reject</button>
          </div>
        </div>
      `;
    });

    deposits.forEach(d => {
      const userName = (allUsersList.find(u => u.uid === d.user_id) || {}).name || (d.user_email || '').split('@')[0];
      const proofCell = d.proof_url
        ? `<a href="${escapeHtml(d.proof_url)}" target="_blank" style="color:inherit; text-decoration:underline;">View<br>proof</a>`
        : 'No<br>proof';
      html += `
        <div class="payment-req-card">
          <div class="payment-user-info">
            <div class="proof-badge">${proofCell}</div>
            <div class="payment-details">
              <h4>${escapeHtml(userName)}</h4>
              <p>${escapeHtml(d.user_email || '')}</p>
              <div class="payment-sub">$${Number(d.amount || 0).toFixed(2)} · Next-tier unlock deposit</div>
              <p style="margin-top: 2px;">${formatDate(d.requested_at)}</p>
            </div>
          </div>
          <div class="payment-actions">
            <button class="btn-verify" onclick="adminDecideDeposit('${d.id}','approved')">Approve & Unlock</button>
            <button class="btn-reject" onclick="adminDecideDeposit('${d.id}','rejected')">Reject</button>
          </div>
        </div>
      `;
    });

    container.innerHTML = html;
  }

  window.adminRejectJoin = async function (id) {
    if (!confirm('Reject this payment / package join request?')) return;
    await SogoPackages.adminSetUserPackageStatus(id, 'deactivated');
    await Promise.all([refreshPayments(), refreshSubscriptions()]);
  };

  window.adminDecideDeposit = async function (id, decision) {
    const result = await SogoPackages.adminDecidePackageDeposit(id, decision);
    if (!result.ok) { alert(result.message); return; }
    await Promise.all([refreshPayments(), refreshSubscriptions()]);
  };

  const paymentsSearchInputElem = document.getElementById('paymentsSearchInput');
  if (paymentsSearchInputElem) {
    paymentsSearchInputElem.oninput = () => renderPaymentsList();
  }
