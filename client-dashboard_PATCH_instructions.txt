async function loadMyDepositsForPackage(userPackageId) {
  if (!window.SogoPackages || !window.SogoPackages.getMyPackageDeposits || !currentSession) return [];
  const all = await SogoPackages.getMyPackageDeposits(currentSession.uid);
  return all.filter(d => d.user_package_id === userPackageId);
}

function findNextLock(pkg, unlockedCount) {
  const locks = pkg.locks || [];
  return locks.filter(l => l.afterReview > unlockedCount).sort((a, b) => a.afterReview - b.afterReview)[0] || null;
}

function renderDepositBannerHtml(pkg, userPkg, deposits) {
  const total = safeAmount(userPkg.totalProducts);
  const unlocked = userPkg.unlockedCount != null ? safeAmount(userPkg.unlockedCount) : total;
  const completed = safeAmount(userPkg.reviewsCompleted);

  // Everything already unlocked, or user hasn't caught up on already-unlocked
  // reviews yet -> no banner needed.
  if (unlocked >= total) return '';
  if (completed < unlocked) return '';

  const nextLock = findNextLock(pkg, unlocked);
  if (!nextLock) return '';

  const pendingDeposit = (deposits || []).find(d => d.status === 'pending');
  if (pendingDeposit) {
    return `
      <div class="card" style="margin-top:16px; background: rgba(217,160,91,0.08); border-color: rgba(217,160,91,0.25);">
        <h4 style="font-size:14px; font-weight:700; margin-bottom:6px; color:#fff;">Deposit under review</h4>
        <p style="font-size:12px; color: var(--text-muted); line-height:1.6;">
          Your deposit of ${formatMoney(pendingDeposit.amount)} is awaiting admin verification.
          The next products will unlock automatically once it's approved.
        </p>
      </div>
    `;
  }

  const inputId = 'depositProofInput';
  return `
    <div class="card" style="margin-top:16px;">
      <h4 style="font-size:15px; font-weight:700; margin-bottom:8px; color:#fff;">You're ready for the next products 🎉</h4>
      <p style="font-size:12px; color: var(--text-muted); line-height:1.6; margin-bottom:14px;">
        Great work — you've reviewed ${completed} products. To open and review the next one and keep earning commission, just one quick step:
      </p>
      <ol style="font-size:12px; color: var(--text-muted); line-height:1.8; margin: 0 0 14px 18px;">
        <li>Make your next deposit of <strong style="color:#fff;">${formatMoney(nextLock.depositRequired)}</strong> and upload your payment screenshot below.</li>
        <li>Once your payment is confirmed, the next product opens up automatically.</li>
        <li>Open it, submit your review, and your commission is added to your balance right away.</li>
      </ol>
      <div style="background: rgba(0,0,0,0.3); border: 1px solid var(--card-border); border-radius: 10px; padding: 12px 16px; display:flex; justify-content:space-between; align-items:center; margin-bottom:14px;">
        <span style="font-size:12px; color:var(--text-muted);">Amount to deposit</span>
        <span style="font-size:14px; font-weight:800; color: var(--accent-gold);">${formatMoney(nextLock.depositRequired)}</span>
      </div>
      <label class="upload-box" for="${inputId}" style="display:block;">
        <i class="fa-solid fa-cloud-arrow-up" style="font-size:24px;color:var(--accent-gold);margin-bottom:8px;"></i>
        <div style="font-size:12px;font-weight:700;color:#fff;" id="${inputId}Label">CHOOSE PAYMENT SCREENSHOT</div>
        <div style="font-size:10px;color:var(--text-muted);margin-top:4px;">JPG, PNG or WEBP</div>
        <input id="${inputId}" type="file" accept="image/*" onchange="document.getElementById('${inputId}Label').textContent = (this.files[0] ? this.files[0].name : 'CHOOSE PAYMENT SCREENSHOT')">
      </label>
      <button class="btn-gold" style="border-radius:6px;padding:10px 20px;" onclick="submitPackageDeposit(${nextLock.depositRequired})">Submit deposit proof</button>
      <div id="depositUploadMessage" style="font-size:11px;margin-top:10px;"></div>
    </div>
  `;
}

async function submitPackageDeposit(amount) {
  const input = document.getElementById('depositProofInput');
  const msg = document.getElementById('depositUploadMessage');
  const file = input && input.files && input.files[0];
  if (!file || !selectedUserPackageId || !currentSession) return;
  const pkg = availablePackages.find(item => item.id === selectedPackageId);

  try {
    const result = await SogoPackages.requestPackageDeposit(
      currentSession.uid,
      currentSession.email,
      selectedUserPackageId,
      pkg ? pkg.id : selectedPackageId,
      amount,
      file
    );
    if (msg) {
      msg.textContent = result.message || (result.ok ? 'Deposit submitted. Waiting for admin verification.' : 'Upload failed.');
      msg.style.color = result.ok ? '#2ecc71' : '#e74c3c';
    }
    if (result.ok) await renderSelectedPackage();
  } catch (e) {
    if (msg) { msg.textContent = e.message || 'Upload failed.'; msg.style.color = '#e74c3c'; }
  }
}
